### plan_video
title: Plan a video
description: Turn a brief into a storyboard: numbered beats with in/out times, per-beat easing roles and a stagger plan that follows the Motion Master doctrine (entrance eases out, exit eases in, on-screen eases in-out, total stagger under 500ms). Deterministic: the same brief always returns the same storyboard.
inputSchema: {
  "type": "object",
  "properties": {
    "brief": {
      "type": "string",
      "minLength": 1,
      "description": "The brief, e.g. \"15s launch film, name reveal, hold 2s, calm exit\"."
    },
    "duration_seconds": {
      "type": "number",
      "exclusiveMinimum": 0,
      "description": "Target length. Parsed from the brief when omitted."
    },
    "style": {
      "type": "string",
      "enum": [
        "premium",
        "calm",
        "energetic",
        "snappy"
      ],
      "description": "Timing personality. Inferred from the brief when omitted."
    },
    "fps": {
      "type": "integer",
      "exclusiveMinimum": 0,
      "description": "Frames per second (default 30)."
    },
    "width": {
      "type": "integer",
      "exclusiveMinimum": 0,
      "description": "Frame width (default 1920)."
    },
    "height": {
      "type": "integer",
      "exclusiveMinimum": 0,
      "description": "Frame height (default 1080)."
    },
    "element_count": {
      "type": "integer",
      "exclusiveMinimum": 0,
      "description": "How many elements the main group has (\"9 words\"). Parsed from the brief when omitted."
    },
    "hold_seconds": {
      "type": "number",
      "exclusiveMinimum": 0,
      "description": "Length of the readable hold. Parsed from the brief when omitted."
    }
  },
  "required": [
    "brief"
  ],
  "additionalProperties": false,
  "$schema": "http://json-schema.org/draft-07/schema#"
}

### mix_footage
title: Mix real footage with motion graphics
description: Validate a Montage timeline (cuts + overlays) and hand it to the user's own Montage instance over its public API: the timeline is written with PUT /api/v1/projects/{project_id}/timeline against the version the instance reports, and the project is validated afterwards. Motion Master never renders footage itself and there is no hosted render service — Montage renders, on the user's own machine.
inputSchema: {
  "type": "object",
  "properties": {
    "timeline": {
      "type": "object",
      "properties": {
        "cuts": {
          "type": "array"
        },
        "overlays": {
          "type": "array"
        },
        "scenes": {
          "type": "array"
        }
      },
      "additionalProperties": true,
      "description": "A Montage timeline: cuts (video/image with in/out seconds) and overlays."
    },
    "project_id": {
      "type": "string",
      "description": "Target Montage project. Resolved from the instance when omitted."
    },
    "dry_run": {
      "type": "boolean",
      "description": "Validate only; never contact the Montage instance."
    },
    "fps": {
      "type": "integer",
      "exclusiveMinimum": 0,
      "description": "Canvas fps (default 30)."
    },
    "width": {
      "type": "integer",
      "exclusiveMinimum": 0,
      "description": "Canvas width (default 1920)."
    },
    "height": {
      "type": "integer",
      "exclusiveMinimum": 0,
      "description": "Canvas height (default 1080)."
    }
  },
  "required": [
    "timeline"
  ],
  "additionalProperties": false,
  "$schema": "http://json-schema.org/draft-07/schema#"
}
